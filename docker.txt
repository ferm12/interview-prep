      - ./mongo-seed:docker-entrypoint-initdb.d

  backend:
    build:
      context: ./backend
    command: npx nx serve backend
    volumes:
      - ./backend:/app
    environment:
      - DB_URL=mongodb://mongo:27017/test
      - JWT_SECRET_KEY=${JWT_SECRET_KEY}
      - JWT_SECRET_KEY_REFRESH=${JWT_SECRET_KEY_REFRESH}
    ports:
      - "3000:3000"
    depends_on:
      - mongo

    volumes:
      - ./apps/frontend:/app

COPY package*.json ./

# Install dependencies
RUN npm install

# Copy the rest of the application code
COPY . .

# Build the frontend (assuming a React/Vue/Next.js app)
RUN npm run build

# Expose the port the app runs on (change if needed)
